## Step 1: 포탈 — parseCookies 유틸 추가

`monkos_AI/portal/lib/shared/config.js`에 추가:

```javascript
// === Cookie 파싱 유틸 ===
export function parseCookies(cookieHeader) {
  const cookies = {};
  if (!cookieHeader) return cookies;
  cookieHeader.split(';').forEach(pair => {
    const [key, ...vals] = pair.trim().split('=');
    if (key) cookies[key.trim()] = vals.join('=').trim();
  });
  return cookies;
}
```

### 2-2. OAuth 콜백 성공 시 쿠키 세팅

Google/카카오/LINE 각 OAuth 콜백에서 **토큰 생성 직후**, `res.writeHead()` 직전에 Set-Cookie 추가.

현재 코드에서 `generateAuthToken(docId, 30 * 24 * 60 * 60 * 1000)` 호출하는 곳을 찾아서, 그 아래에:

```javascript
// 크로스도메인 세션 쿠키 세팅
res.setHeader('Set-Cookie',
  `monkos_session=${token}; Domain=.monkos.ai; Path=/; Secure; HttpOnly; SameSite=Lax; Max-Age=2592000`
);
```

**주의**: `res.writeHead()` 전에 `setHeader`해야 함. `writeHead` 이후에는 헤더 추가 불가.

**팝업 모드 (postMessage 경유)**:
- 팝업 모드에서는 HTML을 직접 res.end()로 보내고 있음
- 이 경우에도 `res.writeHead(200, ...)` 호출 전에 `res.setHeader('Set-Cookie', ...)` 추가

**비팝업 모드 (리다이렉트)**:
- `res.writeHead(302, { Location: '/' })` 하는 곳에서도 Set-Cookie 추가
- `res.setHeader('Set-Cookie', ...)` 후 `res.writeHead(302, ...)`

### 2-3. verify-cookie 엔드포인트 신설

`auth.js` 하단에 추가:

```javascript
/**
 * GET /api/auth/verify-cookie — 쿠키 기반 세션 검증
 * 크로스도메인 세션 복원용. credentials: include로 호출.
 */
export async function handleVerifyCookie(req, res) {
  const cookies = parseCookies(req.headers.cookie || '');
  const token = cookies['monkos_session'];

  if (!token) {
    res.writeHead(401, { 'Content-Type': 'application/json' });
    return res.end(JSON.stringify({ error: 'no session' }));
  }

  const result = verifyAuthToken(token);
  if (!result) {
    // 만료된 쿠키 삭제
    res.setHeader('Set-Cookie',
      'monkos_session=; Domain=.monkos.ai; Path=/; Max-Age=0');
    res.writeHead(401, { 'Content-Type': 'application/json' });
    return res.end(JSON.stringify({ error: 'invalid token' }));
  }

  try {
    const userDoc = await usersCollection.doc(result.userId).get();
    if (!userDoc.exists) {
      res.writeHead(404, { 'Content-Type': 'application/json' });
      return res.end(JSON.stringify({ error: 'user not found' }));
    }

    const userData = userDoc.data();
    res.writeHead(200, { 'Content-Type': 'application/json' });
    res.end(JSON.stringify({
      user: {
        id: result.userId,
        email: userData.email,
        name: userData.name,
        picture: userData.picture,
        credits: userData.credits ?? 0,
      }
    }));
  } catch (err) {
    log(`[Auth] verify-cookie 에러: ${err.message}`);
    res.writeHead(500, { 'Content-Type': 'application/json' });
    res.end(JSON.stringify({ error: err.message }));
  }
}
```

export 목록에 `handleVerifyCookie` 추가 확인.

## Step 6: 앱 — shared.js initCredits() 보강

`monkos_AI/idphoto-kr/public/shared.js`의 `initCredits()` 함수:

**기존 코드 유지 + 쿠키 폴백 추가.** 함수를 `async`로 변경:

```javascript
export async function initCredits() {
  // 1차: sessionStorage 확인 (기존 로직 유지)
  const stored = sessionStorage.getItem('monkos_user');
  if (stored) {
    try {
      const user = JSON.parse(stored);
      if (user.id) {
        store.user = {
          id: user.id,
          email: user.email,
          name: user.name || user.email?.split('@')[0] || '',
          credits: user.credits ?? 0,
          avatar: user.avatar || user.picture || '',
        };
        Monk.log(`로그인: ${user.email} (${store.user.credits}c)`);
        fetchLatestCredits(); // 서버 최신값 동기화
        return;
      }
    } catch (e) {
      Monk.log('사용자 정보 파싱 실패');
    }
  }

  // 2차: 쿠키 세션 복원 시도 (신규 — 크로스도메인 지원)
  try {
    const res = await fetch(`${API.PORTAL_URL}/api/auth/verify-cookie`, {
      credentials: 'include',
    });
    if (res.ok) {
      const data = await res.json();
      if (data.user?.id) {
        store.user = {
          id: data.user.id,
          email: data.user.email,
          name: data.user.name || '',
          credits: data.user.credits ?? 0,
          avatar: data.user.picture || '',
        };
        sessionStorage.setItem('monkos_user', JSON.stringify(data.user));
        Monk.log(`쿠키 세션 복원: ${data.user.email} (${data.user.credits}c)`);
        return;
      }
    }
  } catch (e) {
    Monk.log('쿠키 세션 복원 실패 (무시)');
  }

  Monk.log('비로그인 상태');
}
```

**주의**: `initCredits`가 async로 바뀌므로, 호출하는 곳에서 `await` 처리 필요.
