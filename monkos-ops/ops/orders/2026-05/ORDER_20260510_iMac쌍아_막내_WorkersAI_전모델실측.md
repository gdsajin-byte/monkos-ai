  -H "Authorization: Bearer ${CF_API_TOKEN}" | jq '.result[] | {id, name, description}'

curl -s "https://api.cloudflare.com/client/v4/accounts/${CF_ACCOUNT_ID}/ai/models/search?search=moonshot" \
