### 2-C. inline script 전면 리팩토링

기존 투터치 결제 로직 → 위젯 기반 플로우로 교체.

**핵심 흐름**:
1. 패키지/스테퍼 탭 → `selectPackage(amount, credits, label)` 호출
2. `selectPackage()` → 위젯 영역 노출 + `widgets.setAmount()` + 요약 업데이트
3. "결제하기" 클릭 → `widgets.requestPayment()` 호출

**전체 inline script 교체**:

```html
<script type="module">
import { store, Monk, sharedInit, navigateTo, CHARGE_OPTIONS, KRW_PER_CREDIT, TOSS_CLIENT_KEY } from './shared.js';
import { renderChargeCards } from './modules/charge.js';
import { initChatbot } from './modules/chatbot.js';

const STEP_UNIT = 10;
const STEP_PRICE = 1450;
const STEP_MIN = 1;
const STEP_MAX = 10;

let stepperQty = 1;
let selectedIdx = -1;       // 0~3: 패키지, 4: 스테퍼
let paymentWidgets = null;   // v2 위젯 인스턴스

function fmt(n) { return n.toLocaleString('ko-KR'); }

// ─── 위젯 초기화 (1회) ───
function initWidget() {
  if (paymentWidgets || !store.user.id || !TOSS_CLIENT_KEY) return;

  const tossPayments = TossPayments(TOSS_CLIENT_KEY);
  paymentWidgets = tossPayments.widgets({
    customerKey: store.user.id,   // Firebase Auth uid
  });
}

// ─── 패키지/스테퍼 선택 → 위젯 렌더 ───
async function selectPackage(amount, credits, label) {
  initWidget();

  if (!paymentWidgets) {
    alert('로그인 후 이용 가능합니다.');
    return;
  }

  // 위젯 영역 표시
  const section = document.getElementById('widget-section');
  section.style.display = 'block';

  // 선택 요약 업데이트
  document.getElementById('selected-label').textContent = `${credits}c`;
  document.getElementById('selected-price').textContent = `₩${fmt(amount)}`;
  document.getElementById('pay-btn').textContent = `₩${fmt(amount)} 결제하기`;

  // 위젯 금액 설정 + 렌더
  await paymentWidgets.setAmount({ currency: 'KRW', value: amount });

  // 최초 렌더 또는 금액 변경 시
  const methodEl = document.getElementById('payment-method');
  if (!methodEl.hasChildNodes()) {
    await paymentWidgets.renderPaymentMethods({
      selector: '#payment-method',
      variantKey: 'DEFAULT',
    });
    await paymentWidgets.renderAgreement({
      selector: '#agreement',
      variantKey: 'AGREEMENT',
    });
  }
}

// ─── 결제 실행 ───
async function handlePay() {
  if (!paymentWidgets || selectedIdx < 0) return;

  let amount, credits;
  if (selectedIdx <= 3) {
    const opt = CHARGE_OPTIONS[selectedIdx];
    amount = opt.price;
    credits = opt.credits + opt.bonus;
  } else {
    amount = stepperQty * STEP_PRICE;
    credits = stepperQty * STEP_UNIT;
  }

  const orderId = `MONKOS_${store.user.id.slice(0, 8)}_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`;

  // return 파라미터 보존 (다른 페이지에서 온 경우)
  const returnParam = new URLSearchParams(location.search).get('return');
  const baseUrl = `${location.origin}/charge.html?toss=success`;
  const successUrl = returnParam
    ? `${baseUrl}&return=${encodeURIComponent(returnParam)}`
    : baseUrl;
  const failBase = `${location.origin}/charge.html?toss=fail`;
  const failUrl = returnParam
    ? `${failBase}&return=${encodeURIComponent(returnParam)}`
    : failBase;

  try {
    await paymentWidgets.requestPayment({
      orderId,
      orderName: `MONKOS ${credits} 크레딧`,
      successUrl,
      failUrl,
      customerEmail: store.user.email || undefined,
      customerName: store.user.name || undefined,
    });
  } catch (err) {
    if (err.code === 'PAY_PROCESS_CANCELED' || err.code === 'USER_CANCEL') {
      Monk.log('결제 취소');
    } else {
      alert('결제 오류: ' + (err.message || err.code));
    }
  }
}

// ─── 패키지 카드 렌더 + 이벤트 ───
function renderPackages() {
  const container = document.getElementById('pkg-container');
  if (!container) return;
  container.innerHTML = renderChargeCards(CHARGE_OPTIONS);

  // 단일 탭 → 선택 + 위젯 렌더
  container.querySelectorAll('[data-ci]').forEach(card => {
    card.addEventListener('click', (e) => {
      e.stopPropagation();
      const idx = parseInt(card.dataset.ci);
      const opt = CHARGE_OPTIONS[idx];
      selectedIdx = idx;
      highlightCard(idx);
      selectPackage(opt.price, opt.credits + opt.bonus, `${opt.credits}c +${opt.bonus}c`);
    });
  });
}

// ─── 카드 하이라이트 ───
function highlightCard(activeIdx) {
  document.querySelectorAll('[data-ci]').forEach(card => {
    const i = parseInt(card.dataset.ci);
    const opt = CHARGE_OPTIONS[i];
    if (!opt) return;
    const priceBlock = card.querySelector('.cm-price-block');
    if (!priceBlock) return;

    if (i === activeIdx) {
      priceBlock.style.background = 'linear-gradient(135deg,rgba(34,197,94,0.4),rgba(34,197,94,0.2))';
      priceBlock.innerHTML = '<div style="color:#f8fafc;font-weight:700;font-size:1.1rem;">✓ 선택됨</div>';
    } else {
      const totalC = opt.credits + opt.bonus;
      const strikePrice = totalC * KRW_PER_CREDIT;
      priceBlock.style.background = opt.best
        ? 'linear-gradient(135deg,rgba(245,158,11,0.3),rgba(245,158,11,0.15))'
        : 'linear-gradient(135deg,rgba(167,139,250,0.3),rgba(129,140,248,0.2))';
      priceBlock.innerHTML =
        '<div style="text-decoration:line-through;color:#666;font-size:0.78rem;">₩' + fmt(strikePrice) + '</div>' +
        '<div style="color:#f8fafc;font-weight:700;font-size:1.1rem;">₩' + fmt(opt.price) + '</div>';
    }
  });

  // 스테퍼 하이라이트 해제
  const stepperPay = document.getElementById('stepper-pay');
  if (activeIdx !== 4 && stepperPay) {
    stepperPay.style.background = 'linear-gradient(135deg,rgba(167,139,250,0.3),rgba(129,140,248,0.2))';
  }
}

// ─── 스테퍼 ───
function updateStepper() {
  const qtyEl = document.getElementById('stepper-qty');
  const priceEl = document.getElementById('stepper-price');
  const minBtn = document.getElementById('stepper-minus');
  const maxBtn = document.getElementById('stepper-plus');
  if (qtyEl) qtyEl.textContent = `${stepperQty * STEP_UNIT}c`;
  if (priceEl) priceEl.textContent = `₩${fmt(stepperQty * STEP_PRICE)}`;
  if (minBtn) minBtn.style.opacity = stepperQty <= STEP_MIN ? '0.3' : '1';
  if (maxBtn) maxBtn.style.opacity = stepperQty >= STEP_MAX ? '0.3' : '1';
}

document.getElementById('stepper-pay').addEventListener('click', (e) => {
  e.stopPropagation();
  selectedIdx = 4;
  highlightCard(-1); // 패키지 하이라이트 해제
  const stepperPay = document.getElementById('stepper-pay');
  stepperPay.style.background = 'linear-gradient(135deg,rgba(34,197,94,0.4),rgba(34,197,94,0.2))';
  selectPackage(stepperQty * STEP_PRICE, stepperQty * STEP_UNIT, `${stepperQty * STEP_UNIT}c`);
});

document.getElementById('stepper-minus').addEventListener('click', (e) => {
  e.stopPropagation();
  if (stepperQty > STEP_MIN) stepperQty--;
  updateStepper();
  if (selectedIdx === 4) {
    selectPackage(stepperQty * STEP_PRICE, stepperQty * STEP_UNIT, `${stepperQty * STEP_UNIT}c`);
  }
});

document.getElementById('stepper-plus').addEventListener('click', (e) => {
  e.stopPropagation();
  if (stepperQty < STEP_MAX) stepperQty++;
  updateStepper();
  if (selectedIdx === 4) {
    selectPackage(stepperQty * STEP_PRICE, stepperQty * STEP_UNIT, `${stepperQty * STEP_UNIT}c`);
  }
});

// ─── 결제하기 버튼 ───
document.getElementById('pay-btn').addEventListener('click', handlePay);

// ─── 돌아가기 ───
function goBack() {
  const params = new URLSearchParams(location.search);
  const returnPage = params.get('return') || 'index';
  navigateTo(`${returnPage}.html`);
}
window.goBack = goBack;

// ─── 초기화 ───
async function init() {
  initChatbot();
  await sharedInit();
  renderPackages();
  updateStepper();
  Monk.log('충전 페이지 초기화 완료 (v2 위젯)');
}

init();
</script>
```

### 3-B. initFromTossResult() — return 파라미터 보존 (L187~228)

`history.replaceState` 시 return 파라미터 유지:

```js
export async function initFromTossResult() {
  const params = new URLSearchParams(window.location.search);
  const returnParam = params.get('return');  // ← 추가

  if (params.get('toss') === 'fail') {
    const msg = params.get('message') || '결제 실패';
    // return 보존
    const cleanUrl = returnParam
      ? `${window.location.pathname}?return=${encodeURIComponent(returnParam)}`
      : window.location.pathname;
    history.replaceState({}, '', cleanUrl);
    alert(msg);
    return;
  }

  if (params.get('toss') === 'success') {
    _showTossLoading();

    const paymentKey = params.get('paymentKey');
    const orderId = params.get('orderId');
    const amount = params.get('amount');

    // return 보존
    const cleanUrl = returnParam
      ? `${window.location.pathname}?return=${encodeURIComponent(returnParam)}`
      : window.location.pathname;
    history.replaceState({}, '', cleanUrl);

    if (!paymentKey || !orderId || !amount) { _hideTossLoading(); return; }

    try {
      const res = await fetchWithTimeout(`${API.PORTAL_URL}/api/payments/toss/confirm`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ paymentKey, orderId, amount: Number(amount), userId: store.user.id }),
      }, 15000);
      const data = await res.json();
      _hideTossLoading();
      if (data.success) {
        store.user = { ...store.user, credits: data.credits };
        Monk.log(`결제 완료: ${data.credits}c`);
        alert(`결제가 완료되었습니다!\n현재 크레딧: ${data.credits}c`);
      } else {
        alert('결제 확인 실패: ' + (data.error || ''));
      }
    } catch (err) {
      _hideTossLoading();
      Monk.log('결제 확인 에러: ' + err.message);
    }
  }
}
```
