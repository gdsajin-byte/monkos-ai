## Task 3: 우측 사이드바 — 이미지 저장소

### 3-1. 백엔드: 이미지 목록 API 신규

대상: `monkos_AI/portal/lib/cms/upload.js` + `server.js`

**신규 API**: `GET /api/admin/images`

```javascript
export async function handleImageList(req, res) {
  const admin = requireAdmin(req, res);
  if (!admin) return;

  try {
    const bucket = storage.bucket(GCS_BUCKET);
    const [files] = await bucket.getFiles({ prefix: 'blog/', maxResults: 50 });

    const images = files
      .filter(f => /\.(jpg|jpeg|png|webp|gif)$/i.test(f.name))
      .sort((a, b) => new Date(b.metadata.timeCreated) - new Date(a.metadata.timeCreated))
      .map(f => ({
        name: f.name.replace('blog/', ''),
        url: `https://storage.googleapis.com/${GCS_BUCKET}/${f.name}`,
        size: parseInt(f.metadata.size, 10),
        created: f.metadata.timeCreated,
      }));

    res.writeHead(200, { 'Content-Type': 'application/json' });
    res.end(JSON.stringify({ images }));
  } catch (err) {
    res.writeHead(500, { 'Content-Type': 'application/json' });
    res.end(JSON.stringify({ error: err.message }));
  }
}
```

**server.js 라우팅 추가:**
```javascript
if (urlPath === '/api/admin/images' && req.method === 'GET') return handleImageList(req, res);
```

```javascript
async function loadImageLibrary() {
  const container = document.getElementById('imageLibrary');
  container.innerHTML = '<div style="grid-column:1/-1;text-align:center;color:var(--text-dim);font-size:12px;padding:20px">불러오는 중...</div>';
  try {
    const r = await fetch('/api/admin/images', { credentials: 'include' });
    const { images } = await r.json();
    if (!images || images.length === 0) {
      container.innerHTML = '<div style="grid-column:1/-1;text-align:center;color:var(--text-dim);font-size:12px;padding:20px">업로드된 이미지가 없습니다</div>';
      return;
    }
    container.innerHTML = images.map(img => `
      <div class="img-thumb" onclick="insertImage('${img.url}', '${img.name}')" title="${img.name}">
        <img src="${img.url}" alt="${img.name}" loading="lazy">
      </div>
    `).join('');
  } catch (err) {
    container.innerHTML = '<div style="grid-column:1/-1;color:var(--danger);font-size:12px;padding:20px">로드 실패</div>';
  }
}
```

**클릭→에디터 삽입:**
```javascript
function insertImage(url, name) {
  editor.exec('addImage', { imageUrl: url, altText: name || 'image' });
  showToast('이미지 삽입 완료');
}
```

**사이드바에서 직접 업로드:**
```javascript
async function uploadToLibrary() {
  const fileInput = document.getElementById('imgLibFile');
  fileInput.click();
  fileInput.onchange = async () => {
    const files = fileInput.files;
    if (!files.length) return;
    for (const file of files) {
      const form = new FormData();
      form.append('image', file, file.name);
      await fetch('/api/admin/upload', { method: 'POST', credentials: 'include', body: form });
    }
    showToast(`${files.length}건 업로드 완료`);
    loadImageLibrary();  // 목록 새로고침
    fileInput.value = '';
  };
}
```
