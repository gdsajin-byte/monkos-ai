### 2. `monkos_AI/portal/lib/payments.js` 수정

L3 import에 `LEMONSQUEEZY_WEBHOOK_SECRET_GLOBAL`, `LEMONSQUEEZY_STORE_ID_JP` 추가.

L78~90 webhook 서명 검증을 듀얼 시크릿 방식으로 교체:

```js
// 서명 검증 — 글로벌/JP 듀얼 시크릿
const signature = req.headers['x-signature'];
if (signature) {
  const secrets = [
    LEMONSQUEEZY_WEBHOOK_SECRET_GLOBAL,
    LEMONSQUEEZY_WEBHOOK_SECRET,
  ].filter(Boolean);

  const verified = secrets.some(secret => {
    const hmac = createHmac('sha256', secret);
    hmac.update(body);
    return hmac.digest('hex') === signature;
  });

  if (!verified) {
    log('[Webhook] 서명 검증 실패 (글로벌/JP 모두 불일치)');
    res.writeHead(401, { 'Content-Type': 'application/json' });
    res.end(JSON.stringify({ error: '서명 불일치' }));
    return;
  }
}
```

> 로직: 글로벌 시크릿 먼저 시도 → 실패 시 JP 시크릿 시도 → 둘 다 실패 시 401. timing-safe 비교가 이상적이나, 현재 기존 코드도 `!==` 비교이므로 동일 수준 유지. 추후 보강 가능.
